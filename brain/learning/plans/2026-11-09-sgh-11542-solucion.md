### 2.2 `app/Services/Modules/App/ParametrosGenerales/ParametroGeneralService.php`
```php
<?php

namespace SGH\Services\Modules\App\ParametrosGenerales;

use SGH\Contracts\Services\Modules\App\ParametrosGenerales\IParametroGeneralService;
use SGH\Models\ParametroGeneral;
use SGH\Services\BaseService;

/**
 * Lee e interpreta los parámetros generales guardados como texto o JSON.
 */
class ParametroGeneralService extends BaseService implements IParametroGeneralService
{
    const UNIDADES_NEGOCIO = 'PPTO_UNIDADES_NEGOCIO';

    public function __construct(ParametroGeneral $parametroGeneral)
    {
        parent::__construct($parametroGeneral);
    }

    public function obtenerEntradas($clave)
    {
        $entradas = json_decode((string) getGlobalParameter($clave), true);

        return is_array($entradas) ? $this->minusculas($entradas) : [];
    }

    public function obtenerPorEmpresa($clave, $emplId)
    {
        foreach ($this->obtenerEntradas($clave) as $entrada) {
            if ((int) data_get($entrada, 'empresa', -1) === (int) $emplId) {
                return $entrada;
            }
        }

        return null;
    }

    public function obtenerUnidadesNegocio($emplId)
    {
        $unidades = [];
        if (is_null($emplId) || $emplId === '') {
            return $unidades;
        }
        $entrada = $this->obtenerPorEmpresa(self::UNIDADES_NEGOCIO, $emplId);
        foreach ((array) data_get($entrada, 'unidades', []) as $unidad) {
            $codigo = (string) data_get($unidad, 'codigo');
            $unidades[$codigo] = $codigo . ' - ' . data_get($unidad, 'descripcion');
        }

        return $unidades;
    }

    public function validarValor($clave, $valor)
    {
        $valor = trim((string) $valor);
        if (!in_array(substr($valor, 0, 1), ['[', '{'])) {
            return preg_match('/[\r\n]/', $valor) ? 'El valor no puede tener saltos de línea.' : null;
        }
        $entradas = json_decode($valor, true);
        if (json_last_error() !== JSON_ERROR_NONE) {
            return 'El valor no es un JSON válido: ' . json_last_error_msg() . '.';
        }

        return $clave === self::UNIDADES_NEGOCIO ? $this->validarUnidades($this->minusculas($entradas)) : null;
    }

    /**
     * Revisa que cada entrada tenga empresa y unidades con código de 001 a 999 y descripción.
     *
     * @param array $entradas
     * @return string|null
     */
    private function validarUnidades(array $entradas)
    {
        foreach ($entradas as $entrada) {
            if (!is_numeric(data_get($entrada, 'empresa')) || !is_array(data_get($entrada, 'unidades'))) {
                return 'Cada entrada debe tener "empresa" y "unidades".';
            }
            foreach ($entrada['unidades'] as $unidad) {
                $codigo = (string) data_get($unidad, 'codigo');
                if (!preg_match('/^\d{3}$/', $codigo) || $codigo === '000' || !data_get($unidad, 'descripcion')) {
                    return 'La unidad "' . $codigo . '" de la empresa ' . $entrada['empresa'] . ' debe tener código de 001 a 999 y descripción.';
                }
            }
        }

        return null;
    }

    /**
     * Pasa a minúsculas las claves del arreglo en todos sus niveles.
     *
     * @param array $datos
     * @return array
     */
    private function minusculas(array $datos)
    {
        $datos = array_change_key_case($datos, CASE_LOWER);
        foreach ($datos as $clave => $valor) {
            if (is_array($valor)) {
                $datos[$clave] = $this->minusculas($valor);
            }
        }

        return $datos;
    }
}
```
Nota: los métodos públicos ya tienen su docblock en el contrato; en la clase se repite el bloque de 1 renglón + `@param`/`@return` (regla del cerebro: toda función documentada). Escribir los mismos textos del contrato.
Nota PHP: una clave `"100"` se vuelve entero en el arreglo; no afecta `array_key_exists('100', …)` ni el `<option value>`.
